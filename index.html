Private Function DeleteRowsByValue(ByVal ws As Worksheet, ByVal target As String, ByVal column As Long) As Long
    'column列の値がtargetの行を削除する
    '削除した行数を返す
    Dim v As Variant
    Dim i As Long
    Dim lastRow As Long
    Dim deleteRowRng As Range

    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    v = ws.Range(ws.Cells(1, column), ws.Cells(lastRow, column)).value

    For i = 1 To lastRow
        If v(i, 1) = target Then
            If deleteRowRng Is Nothing Then
                Set deleteRowRng = ws.Rows(i)
            Else
                Set deleteRowRng = Union(deleteRowRng, ws.Rows(i))
            End If
        End If
    Next i

    If Not deleteRowRng Is Nothing Then
        DeleteRowsByValue = deleteRowRng.Rows.Count
        deleteRowRng.Delete
    Else
        DeleteRowsByValue = 0
    End If
End Function

Private Function DeleteRowsExceptValue(ByVal ws As Worksheet, ByVal keepValue As String, ByVal columnIndex As Long) As Long
    'columnIndex 列の値が keepValue 以外の行を削除する
    '削除した行数を返す
    
    Dim v As Variant
    Dim i As Long
    Dim lastRow As Long
    Dim deleteRowRng As Range

    lastRow = ws.Cells(ws.Rows.Count, columnIndex).End(xlUp).Row

    ' 対象列を配列化（高速）
    v = ws.Range(ws.Cells(1, columnIndex), ws.Cells(lastRow, columnIndex)).value

    ' keepValue 以外の行を削除対象に追加
    For i = 1 To lastRow
        If v(i, 1) <> keepValue Then
            If deleteRowRng Is Nothing Then
                Set deleteRowRng = ws.Rows(i)
            Else
                Set deleteRowRng = Union(deleteRowRng, ws.Rows(i))
            End If
        End If
    Next i

    ' 削除実行
    If Not deleteRowRng Is Nothing Then
        DeleteRowsExceptValue = deleteRowRng.Rows.Count
        deleteRowRng.Delete
    Else
        DeleteRowsExceptValue = 0
    End If
End Function

Private Function HasTimeProblem(ByRef msg As String) As Boolean
    msg = ""                    ' メッセージ初期化
    HasTimeProblem = False      ' 初期値

    If Now > TimeValue("15:00:00") Then
        msg = msg & "15時以降です。" & vbCrLf
    End If

    If Now < TimeValue("20:00:00") Then
        msg = msg & "20時以前です。" & vbCrLf
    End If

    If msg <> "" Then
        HasTimeProblem = True
    End If
End Function


Sub ExecuteProcess()
    Dim message As String

    If HasTimeProblem(message) Then
        MsgBox message & vbCrLf & "時間の問題があります。処理を終了します。"
        Exit Sub
    End If
End Sub

Private Function CollectionToArray(ByVal collection As collection, ByVal startIndex As Long) As Variant()
    'コレクションを配列に変換する
    Dim arr() As Variant
    Dim i As Long

    ' 配列を開始番号に合わせて確保
    ReDim arr(startIndex To startIndex + collection.Count - 1)

    ' Collection → 配列へコピー
    For i = 1 To collection.Count
        arr(startIndex + i - 1) = collection.Item(i)
    Next i

    CollectionToArray = arr
End Function

Private Function IsWorkbookLocked(ByVal filePath As String) As Boolean
    'ブックが開ひらかれているか調べる
    Dim fileNum As Integer

    On Error GoTo Locked

    fileNum = FreeFile

    ' 追記モード（書き込みロック）で開けるか試す
    Open filePath For Binary Access Write Lock Read Write As #fileNum

    ' 開けた → ロックされていない
    Close #fileNum
    IsWorkbookLocked = False
    Exit Function

Locked:
    ' 開けない → 誰かが開いている（編集中）
    IsWorkbookLocked = True
End Function

Private Function GetLastRow(ByVal ws As Worksheet) As Long
    '最終行を取得
    Dim lastRow As Long
    Dim col_i As Long
    Dim r As Long

    lastRow = 0

    ' 実務で使う範囲だけ（1～100列）
    For col_i = 1 To 100
        r = ws.Cells(ws.Rows.Count, col_i).End(xlUp).Row
        If r > lastRow Then lastRow = r
    Next col_i

    GetLastRow = lastRow
End Function

Private Function HasDuplicate(ByVal ws As Worksheet, ByVal checkColumn As Integer) As Boolean
    '重複があるか調べる
    Dim dic As Object
    Dim v As Variant
    Dim i As Long
    Dim lastRow As Long
    
    HasDuplicate = False

    Set dic = CreateObject("Scripting.Dictionary")
    
    lastRow = ws.Cells(ws.Rows.Count, checkColumn).End(xlUp).Row
    v = ws.Range(ws.Cells(1, checkColumn), ws.Cells(lastRow, checkColumn)).value

    For i = 1 To UBound(v)
        If dic.Exists(v(i, 1)) Then
            HasDuplicate = True
            Exit Function
        Else
            dic.Add v(i, 1), i
        End If
    Next i
    
End Function

Private Function HasDuplicate(ByVal ws As Worksheet, ByVal checkColumn1 As Integer, ByVal checkColumn2 As Integer, ByVal checkColumn3 As Integer) As Boolean
    '重複があるか調べる
    Dim dic As Object
    Dim v As Variant
    Dim i As Long
    Dim lastRow As Long
    Dim maxColumn As Long
    Dim minColumn As Long
    Dim key As String
    
    HasDuplicate = False

    Set dic = CreateObject("Scripting.Dictionary")
    
    lastRow = ws.Cells(ws.Rows.Count, checkColumn1).End(xlUp).Row
    maxColumn = WorksheetFunction.Max(checkColumn1, checkColumn2, checkColumn3)
    minColumn = WorksheetFunction.Min(checkColumn1, checkColumn2, checkColumn3)
    
    v = ws.Range(ws.Cells(1, minColumn), ws.Cells(lastRow, maxColumn)).value

    For i = 1 To UBound(v)
        key = CStr(v(i, checkColumn1)) & "|" & CStr(v(i, checkColumn2)) & "|" & CStr(v(i, checkColumn3))
        If dic.Exists(key) Then
            HasDuplicate = True
            Exit Function
        Else
            dic.Add key, i
        End If
    Next i
    
End Function

Private Function RenameFile(ByVal folderPath As String, ByVal oldName As String, ByVal newName As String) As Boolean
    'ファイル名を変更する
    
    Dim oldPath As String
    Dim newPath As String
    
    RenameFile = False  '失敗を初期値にする
    
    ' フルパスを作成
    oldPath = folderPath & "\" & oldName
    newPath = folderPath & "\" & newName
    
    ' 元ファイルが存在するか確認
    If Dir(oldPath) = "" Then
        Exit Function   '存在しない → False のまま終了
    End If
    
    ' 変更後ファイル名が既に存在する場合は失敗扱い
    If Dir(newPath) <> "" Then
        Exit Function
    End If
    
    ' ファイル名変更
    Name oldPath As newPath
    
    RenameFile = True   '成功
End Function

Private Function SafeMid(ByVal txt As Variant, _
                        ByVal startPos As Long, _
                        ByVal length As Long) As String
    ' WorksheetFunction.Mid の代わりに使う安全版
    
    ' 空・Null・非文字列なら空文字を返す
    If IsNull(txt) Or IsEmpty(txt) Then
        SafeMid = ""
        Exit Function
    End If
    
    Dim s As String
    s = CStr(txt)
    
    ' 開始位置が不正なら空文字
    If startPos < 1 Or startPos > Len(s) Then
        SafeMid = ""
        Exit Function
    End If
    
    ' 長さが不正なら空文字
    If length < 1 Then
        SafeMid = ""
        Exit Function
    End If
    
    ' 安全に Mid を返す
    SafeMid = Mid$(s, startPos, length)
End Function

Private Function ExcludeSet(ByVal setA As Variant, ByVal setB As Variant) As Variant
    '1次元配列の配列setAの中から、1次元配列の配列setBに含まれる値を除外する
    'Rangeで取得した配列は１行でも２次元配列になるためこの関数は使えない
    '例
    'A = Array("A", "B", "C", "D", "E")
    'B = Array("B", "D")
    'C = ExcludeSet(A, B)
    'for文で出力すると,A,C,Eが出力される
    
    Dim dic As Object
    Dim result() As Variant
    Dim i As Long, cnt As Long
    
    Set dic = CreateObject("Scripting.Dictionary")
    
    ' --- 集合BをDictionaryに登録（高速検索用） ---
    For i = LBound(setB) To UBound(setB)
        If Not dic.Exists(CStr(setB(i))) Then
            dic.Add CStr(setB(i)), True
        End If
    Next i
    
    ' --- 結果配列を作成 ---
    ReDim result(0 To UBound(setA))
    cnt = 0
    
    ' --- 集合Aの値を走査し、集合Bにないものだけ追加 ---
    For i = LBound(setA) To UBound(setA)
        If Not dic.Exists(CStr(setA(i))) Then
            result(cnt) = setA(i)
            cnt = cnt + 1
        End If
    Next i
    
    ' --- 実際の件数に合わせて配列を縮める ---
    ReDim Preserve result(0 To cnt - 1)
    
    ExcludeSet = result
End Function

Private Function ExcludeRange(ByVal rngA As Range, ByVal rngB As Range) As Variant
    '1列の2次元配列の配列setAの中から、1列の2次元配列の配列setBに含まれる値を除外する
    'Rangeで取得した配列しか使えない
    '例
    'set A = ws.Range("A1:A10")
    'set B = ws.Range("B1:B10")
    'C = ExcludeRange(A, B)

    Dim dic As Object
    Dim arrA As Variant, arrB As Variant
    Dim result() As Variant
    Dim i As Long, cnt As Long
    Dim rowsA As Long, rowsB As Long
    
    Set dic = CreateObject("Scripting.Dictionary")
    
    rowsA = rngA.Rows.Count
    rowsB = rngB.Rows.Count
    
    arrA = rngA.value
    arrB = rngB.value
    
    ' 集合BをDictionaryに登録
    For i = 1 To rowsB
        If Not dic.Exists(CStr(arrB(i, 1))) Then
            dic.Add CStr(arrB(i, 1)), True
        End If
    Next i
    
    ' 結果配列（1次元）を最大サイズで確保
    ReDim result(1 To rowsA)
    cnt = 0
    
    ' 集合Aの値を走査し、集合Bにないものだけ追加
    For i = 1 To rowsA
        If Not dic.Exists(CStr(arrA(i, 1))) Then
            cnt = cnt + 1
            result(cnt) = arrA(i, 1)
        End If
    Next i
    
    ' 実際の件数に合わせて縮める
    If cnt = 0 Then
        ExcludeRange = Array()   '空集合
    Else
        ReDim Preserve result(1 To cnt)
        ExcludeRange = result
    End If
End Function

Private Function IntersectArrayUnique(ByVal arrA As Variant, ByVal arrB As Variant) As Variant
    '1次元配列の配列arrAの中に含まれる値かつ、1次元配列の配列arrBにも含まれる値を取得する
    'AかつBの同じ値が複数ある場合は一つのみ取得する
    'Rangeで取得した配列は１行でも２次元配列になるためこの関数は使えない
    '例
    'A = Array("A", "B", "C", "D", "E", "D")
    'B = Array("B", "D")
    'C = IntersectRangeUnique(A, B)
    'for文で出力すると,B,Dが出力される
    
    Dim dicB As Object
    Dim dicResult As Object
    Dim result() As Variant
    Dim i As Long, cnt As Long
    
    Set dicB = CreateObject("Scripting.Dictionary")
    Set dicResult = CreateObject("Scripting.Dictionary")
    
    ' --- 集合BをDictionaryに登録（重複除去） ---
    For i = LBound(arrB) To UBound(arrB)
        dicB(CStr(arrB(i))) = True
    Next i
    
    ' --- 結果配列（最大サイズで確保） ---
    ReDim result(LBound(arrA) To UBound(arrA))
    cnt = LBound(arrA)
    
    ' --- 集合Aの値が集合Bに存在し、かつ重複していない場合だけ追加 ---
    For i = LBound(arrA) To UBound(arrA)
        Dim key As String
        key = CStr(arrA(i))
        
        If dicB.Exists(key) Then
            If Not dicResult.Exists(key) Then
                result(cnt) = key
                dicResult.Add key, True
                cnt = cnt + 1
            End If
        End If
    Next i
    
    ' --- 実際の件数に合わせて縮める ---
    If dicResult.Count = 0 Then
        IntersectArrayUnique = Array()   '空集合
    Else
        ReDim Preserve result(LBound(arrA) To cnt - 1)
        IntersectArrayUnique = result
    End If
End Function
